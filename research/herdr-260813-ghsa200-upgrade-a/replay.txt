import json
from pathlib import Path
pages = Path("/tmp/ghsa200-worker-clones/upgrade-a/pages")

def load_ok(path: Path):
    d = json.loads(path.read_text())
    assert isinstance(d, dict) and not d.get("error")
    ghsa = (d.get("ghsa_id") or d.get("ghsaId") or d.get("id") or "").lower()
    assert ghsa.startswith("ghsa-")
    assert not (d.get("withdrawn_at") or d.get("withdrawnAt"))
    return ghsa, d.get("cve_id") or d.get("cveId")

pairs = [
    ("ghsa/GHSA-fmfg-9g7c-3vq7.json", "ghsa-fmfg-9g7c-3vq7", "CVE-2026-32111"),
    ("ghsa/GHSA-xw8c-rrvx-f7xq.json", "ghsa-xw8c-rrvx-f7xq", "CVE-2026-44219"),
    ("ghsa/GHSA-wv46-v6xc-2qhf.json", "ghsa-wv46-v6xc-2qhf", "CVE-2026-35670"),
    ("ghsa/GHSA-rg8m-3943-vm6q.json", "ghsa-rg8m-3943-vm6q", "CVE-2026-41376"),
]
for rel, expect, cve in pairs:
    ghsa, got_cve = load_ok(pages / rel)
    assert ghsa == expect, (ghsa, expect)
    assert got_cve == cve, (got_cve, cve)
    print("ok", rel, ghsa, got_cve)
print("first-party GHSA objects PASS")
PY
