===== from_bytes =====
    pub fn from_bytes(bytes: BytesN<32>) -> Self {

    }

impl From<U256> for Fr {
    fn from(value: U256) -> Self {
        Self(value)
    }
}

        self.as_u256().partial_cmp(other.as_u256()) == Some(core::cmp::Ordering::Equal)
    }
}
