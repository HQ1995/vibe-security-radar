      : result.deno_args.map((a) => {
        // POSIX: args with shell variable refs use double quotes to
        // preserve variable expansion. Other metacharacters use single
        // quotes.
        if (/\$\{[^}]+\}|\$[A-Za-z_]/.test(a)) {
          return '"' + a.replace(/"/g, '\\"') + '"';
        }
        if (/[();&|<>`!\n\r\s"'\\$]/.test(a)) {
          return "'" + a.replace(/'/g, "'\\''") + "'";
        }
        return a;
