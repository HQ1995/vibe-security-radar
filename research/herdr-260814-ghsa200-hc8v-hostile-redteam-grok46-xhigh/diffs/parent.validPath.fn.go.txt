514:	for _, p := range paths {
515:		parts := strings.FieldsFunc(p, func(r rune) bool { return (r == '\\' || r == '/') })
516:		if len(parts) == 0 {
517:			return fmt.Errorf("invalid path: %q", p)
518:		}

521:			return fmt.Errorf("invalid path prefix: %q", p)
522:		}

524:		if runtime.GOOS == "windows" {
525:			// Volume names are not supported, in both formats: \\ and <DRIVE_LETTER>:.

552:func windowsValidPath(part string) bool {
553:	if len(part) > 3 && strings.EqualFold(part[:4], GitDirName) {
554:		// For historical reasons, file names that end in spaces or periods are
555:		// automatically trimmed. Therefore, `.git . . ./` is a valid way to refer
556:		// to `.git/`.
557:		if windowsPathReplacer.Replace(part[4:]) == "" {
558:			return false
559:		}
560:
561:		// For yet other historical reasons, NTFS supports so-called "Alternate Data
562:		// Streams", i.e. metadata associated with a given file, referred to via
563:		// `<filename>:<stream-name>:<stream-type>`. There exists a default stream
564:		// type for directories, allowing `.git/` to be accessed via
565:		// `.git::$INDEX_ALLOCATION/`.
566:		//
567:		// For performance reasons, _all_ Alternate Data Streams of `.git/` are
568:		// forbidden, not just `::$INDEX_ALLOCATION`.
569:		if len(part) > 4 && part[4:5] == ":" {
570:			return false
571:		}
572:	}
573:	return true
