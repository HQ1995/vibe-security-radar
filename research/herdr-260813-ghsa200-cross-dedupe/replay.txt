# 7. Git release-containment verification (candidate in vulnerable_tag,
#    fix in fixed_tag, fix not in vulnerable_tag) for every PASS row with a local clone.
#    Clone roots: /tmp/fp211-cross-04/clones, /tmp/ghsa200-worker-clones/*/clones.
python3 - <<'PY'
import json, os, subprocess
ledger = [json.loads(l) for l in open('/home/hanqing/agents/ai-slop/autoresearch/orchestrator-260813-fp211-canonical/ledger.jsonl')]
roots = ['/tmp/fp211-cross-04/clones', '/tmp/ghsa200-worker-clones/upgrade-a/clones',
         '/tmp/ghsa200-worker-clones/fresh-am/clones', '/tmp/ghsa200-worker-clones/remediation']
clones = {}
for root in roots:
    if os.path.isdir(root):
        for n in os.listdir(root):
            p = os.path.join(root, n)
            if os.path.isdir(p) and (os.path.isdir(p + '/.git') or os.path.isdir(p + '/objects')):
                clones[n.lower()] = p

def anc(c, s, t):
    if not s or not t:
        return None
    return subprocess.run(['git', '-C', c, 'merge-base', '--is-ancestor', s, t]).returncode == 0

out = []
for r in ledger:
    if not (r.get('record_kind') == 'COMPONENT_ROW' and r.get('row_state') == 'PASS' and r.get('release_evidence')):
        continue
    repo = r.get('repository')
    slug = repo.lower().split('/')[-1] if repo else None
    c = clones.get(slug)
    if not c:
        continue
    ev = r['release_evidence']
    vuln, fixed = ev.get('vulnerable_tag'), ev.get('fixed_tag')
    if not vuln or not fixed:
        continue
    cand = ev.get('candidate_sha') or ev.get('candidate_member_sha')
    fix = ev.get('fix_sha') or ev.get('minimum_security_closure_sha') or ev.get('fix_member_sha')
    out.append({'row': r['row_key'], 'repo': repo, 'vulnerable_tag': vuln, 'fixed_tag': fixed,
                'candidate_in_vulnerable': anc(c, cand, vuln),
                'fix_in_fixed': anc(c, fix, fixed),
                'fix_in_vulnerable': anc(c, fix, vuln)})
json.dump(out, open('work/git_verify.json', 'w'), indent=2)
print('verified', len(out), 'rows')
PY

# 8. Regenerate the terminal artifacts (deterministic).
