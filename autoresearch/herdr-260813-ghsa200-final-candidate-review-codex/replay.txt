# Run from /home/hanqing/agents/ai-slop.
OUT=autoresearch/herdr-260813-ghsa200-final-candidate-review-codex
CONTRACT=autoresearch/orchestrator-260813-ghsa200-leader/CONTRACT.md

# Verify the contract and the final case ledger binding.
test "$(sha256sum "$CONTRACT" | awk '{print $1}')" = cbd04ef2770d7b93e59032ec92dde3566e1d64df3b1e003dfbcfbea558f90ed3
test "$(sha256sum "$OUT/cases.jsonl" | awk '{print $1}')" = e275437954890dca07855b5fcfa545f8f1a366fb85a7ee9f067da5b710b2b3da

# Verify all frozen core and packet inputs recorded by the manifest.
jq -r '.frozen_core_inputs[] | [.sha256,.path] | @tsv' "$OUT/manifest.json" |
while IFS=$'\t' read -r expected file; do
  test "$(sha256sum "$file" | awk '{print $1}')" = "$expected"
done
jq -r '.frozen_packet_inputs[] | .packet as $packet | .files | to_entries[] | [.value,($packet + "/" + .key)] | @tsv' "$OUT/manifest.json" |
while IFS=$'\t' read -r expected file; do
  test "$(sha256sum "$file" | awk '{print $1}')" = "$expected"
done

# Parse all structured artifacts and prove exact candidate/counting invariants.
jq -e . "$OUT/manifest.json" "$OUT/result.json" >/dev/null
jq -ce . "$OUT/cases.jsonl" >/dev/null
jq -es '
  length == 32 and
  (map(select(.final_verdict == "ACCEPT")) | length) == 31 and
  (map(select(.final_verdict == "NARROW")) | length) == 1 and
  (map(select(.final_verdict == "REJECT")) | length) == 0 and
  (map(select(.final_verdict == "UNKNOWN")) | length) == 0 and
  (map(select(.final_verdict == "BLOCKED")) | length) == 0 and
  (map(select(.countable_first_party_ghsa)) | length) == 31 and
  (map(select(.baseline_relation == "BASELINE_REVALIDATION_NOT_NET_NEW")) | length) == 10 and
  (map(select(.baseline_relation != "BASELINE_REVALIDATION_NOT_NET_NEW")) | length) == 22 and
  (map(.case_id) | unique | length) == 32 and
  (map(.mechanism_key) | unique | length) == 32 and
  all(.[] | select(.final_verdict == "ACCEPT"); all(.gates[]; . == "PASS" or . == "NOT_APPLICABLE")) and
  ([.[] | select(.case_id == "GHSA-4FXP-2M36-QV64")][0].gates.identity_gate == "NARROW")
' "$OUT/cases.jsonl" >/dev/null
jq -e '
  .candidate_counts == {"ACCEPT":31,"NARROW":1,"REJECT":0,"UNKNOWN":0,"BLOCKED":0,"total":32} and
  (.countable_first_party_ghsa_ids | length) == 31 and
  (.countable_first_party_ghsa_ids | unique | length) == 31 and
  .commitfirst_gn_boundary.explicitly_unreviewed == 2476 and
  .commitfirst_gn_boundary.whole_shard_completion_inferred == false
' "$OUT/result.json" >/dev/null

# Re-run the frozen per-case topology/release commands exactly as recorded.
# Each cases.jsonl row names its source packet replay plus any row-specific commands.
jq -r '[.case_id,.source_binding.packet,(.replay_commands | join(" || "))] | @tsv' "$OUT/cases.jsonl"

# English-only/Han, owned-file, and whitespace checks.
test -z "$(rg -n --pcre2 '\p{Han}' "$OUT"/{manifest.json,cases.jsonl,result.json,report.md,replay.txt} || true)"
test "$(find "$OUT" -maxdepth 1 -type f -printf '%f\n' | sort | tr '\n' ' ')" = "cases.jsonl manifest.json replay.txt report.md result.json "
git diff --check -- "$OUT"
